// src/components/ThemeModeSelector.jsx
import React, { useState, useEffect } from 'react';
import { useTheme } from '../contexts/ThemeContext';
import { FiSun, FiMoon, FiMonitor, FiDroplet, FiCheck, FiInfo } from 'react-icons/fi';
import './ThemeModeSelector.css';

const ThemeModeSelector = () => {
  const { themeMode, changeThemeMode, theme } = useTheme();
  const [devicePrefersDark, setDevicePrefersDark] = useState(false);
  const [showInfo, setShowInfo] = useState(false);

  // Check device theme preference
  useEffect(() => {
    const mediaQuery = window.matchMedia('(prefers-color-scheme: dark)');
    setDevicePrefersDark(mediaQuery.matches);
    
    const handleChange = (e) => {
      setDevicePrefersDark(e.matches);
    };
    
    mediaQuery.addEventListener('change', handleChange);
    return () => mediaQuery.removeEventListener('change', handleChange);
  }, []);

  // Theme modes configuration
  const modes = [
    {
      id: 'light',
      name: 'Light',
      icon: <FiSun />,
      description: 'Bright theme optimized for daytime use and well-lit environments',
      features: [
        'White backgrounds with colored accents',
        'Dark text for maximum readability',
        'Reduced eye strain in bright light',
        'Professional corporate look'
      ],
      recommendedFor: 'Daytime work, office environments, presentations'
    },
    {
      id: 'dark',
      name: 'Dark',
      icon: <FiMoon />,
      description: 'Dark theme that reduces eye strain in low-light conditions',
      features: [
        'Dark backgrounds with light text',
        'Reduces blue light exposure',
        'Better for nighttime usage',
        'Modern and sleek appearance'
      ],
      recommendedFor: 'Night work, low-light environments, prolonged usage'
    },
    {
      id: 'device',
      name: 'Device',
      icon: <FiMonitor />,
      description: 'Automatically follows your device theme settings',
      features: [
        `Currently detecting: ${devicePrefersDark ? 'Dark Mode' : 'Light Mode'}`,
        'Switches with system theme changes',
        'Seamless integration with OS',
        'No manual switching needed'
      ],
      recommendedFor: 'Users who prefer system-wide theme synchronization'
    },
    {
      id: 'custom',
      name: 'Custom',
      icon: <FiDroplet />,
      description: 'Create your own color theme with custom color selection',
      features: [
        'Choose any color you like',
        'Automatic shade generation',
        'Full control over appearance',
        'Save and reuse custom themes'
      ],
      recommendedFor: 'Brand customization, personal preference, creative projects'
    }
  ];

  // Get current mode details
  const currentMode = modes.find(mode => mode.id === themeMode) || modes[0];

  // Handle mode change
  const handleModeChange = (modeId) => {
    changeThemeMode(modeId);
  };

  // Get mode status text
  const getModeStatus = (modeId) => {
    if (modeId === 'device') {
      return devicePrefersDark ? '🌙 Dark Mode Detected' : '☀️ Light Mode Detected';
    }
    return '';
  };

  return (
    <div className="theme-mode-selector">
      {/* Header */}
      <div className="selector-header">
        <h3 className="section-title">
          <span className="title-icon">🌓</span>
          Appearance Mode
        </h3>
        <button 
          className="info-button"
          onClick={() => setShowInfo(!showInfo)}
          title="Show mode information"
        >
          <FiInfo />
        </button>
      </div>
      
      <p className="section-description">
        Choose how you want your theme to behave. Each mode offers different benefits for various usage scenarios.
      </p>

      {/* Current Mode Status */}
      <div className="current-mode-status">
        <div className="status-header">
          <span className="status-label">Current Mode:</span>
          <span className="status-value">
            <span className={`mode-badge ${themeMode}`}>
              {currentMode.icon}
              {currentMode.name}
            </span>
          </span>
        </div>
        
        {themeMode === 'device' && (
          <div className="device-status">
            <span className="status-icon">{devicePrefersDark ? '🌙' : '☀️'}</span>
            <span className="status-text">
              Your device is currently in <strong>{devicePrefersDark ? 'Dark' : 'Light'}</strong> mode
            </span>
          </div>
        )}
        
        {themeMode === 'custom' && (
          <div className="custom-status">
            <span className="status-icon" style={{ color: theme.colors.primary }}>🎨</span>
            <span className="status-text">
              Using custom color: <strong>{theme.colors.primary}</strong>
            </span>
          </div>
        )}
      </div>

      {/* Info Panel */}
      {showInfo && (
        <div className="info-panel">
          <div className="info-content">
            <h4>About Appearance Modes</h4>
            <p>
              Appearance modes determine how your theme behaves. You can choose a fixed light or dark theme, 
              sync with your device settings, or create a completely custom color theme.
            </p>
            
            <div className="mode-details">
              <div className="detail-item">
                <span className="detail-icon">💡</span>
                <div className="detail-content">
                  <strong>Light Mode</strong>
                  <p>Best for daytime use, reduces eye strain in bright environments</p>
                </div>
              </div>
              
              <div className="detail-item">
                <span className="detail-icon">🌙</span>
                <div className="detail-content">
                  <strong>Dark Mode</strong>
                  <p>Reduces eye fatigue in low-light conditions, saves battery on OLED screens</p>
                </div>
              </div>
              
              <div className="detail-item">
                <span className="detail-icon">🔄</span>
                <div className="detail-content">
                  <strong>Device Mode</strong>
                  <p>Automatically switches between light/dark based on your system settings</p>
                </div>
              </div>
              
              <div className="detail-item">
                <span className="detail-icon">🎨</span>
                <div className="detail-content">
                  <strong>Custom Mode</strong>
                  <p>Create your own unique color scheme with complete control</p>
                </div>
              </div>
            </div>
          </div>
          
          <button 
            className="close-info-btn"
            onClick={() => setShowInfo(false)}
          >
            Close
          </button>
        </div>
      )}

      {/* Mode Selection Grid */}
      <div className="mode-selection-grid">
        {modes.map((mode) => (
          <div
            key={mode.id}
            className={`mode-card ${themeMode === mode.id ? 'active' : ''} ${mode.id}`}
            onClick={() => handleModeChange(mode.id)}
          >
            {/* Card Header */}
            <div className="mode-card-header">
              <div className="mode-icon-wrapper">
                <div className="mode-icon">
                  {mode.icon}
                </div>
                {themeMode === mode.id && (
                  <div className="active-indicator">
                    <FiCheck className="check-icon" />
                  </div>
                )}
              </div>
              
              <div className="mode-title">
                <h4>{mode.name}</h4>
                <span className="mode-status">{getModeStatus(mode.id)}</span>
              </div>
            </div>
            
            {/* Card Content */}
            <div className="mode-card-content">
              <p className="mode-description">{mode.description}</p>
              
              <div className="mode-features">
                <h5>Key Features:</h5>
                <ul className="features-list">
                  {mode.features.slice(0, 3).map((feature, index) => (
                    <li key={index} className="feature-item">
                      <span className="feature-dot">•</span>
                      <span>{feature}</span>
                    </li>
                  ))}
                </ul>
              </div>
              
              <div className="mode-recommendation">
                <span className="recommendation-label">Best for:</span>
                <span className="recommendation-text">{mode.recommendedFor}</span>
              </div>
            </div>
            
            {/* Card Footer */}
            <div className="mode-card-footer">
              {themeMode === mode.id ? (
                <button className="mode-action-btn active">
                  <FiCheck className="btn-icon" />
                  <span className="btn-text">Currently Active</span>
                </button>
              ) : (
                <button className="mode-action-btn">
                  <span className="btn-icon">→</span>
                  <span className="btn-text">Select This Mode</span>
                </button>
              )}
            </div>
            
            {/* Active Badge */}
            {themeMode === mode.id && (
              <div className="active-badge">
                <div className="badge-content">
                  <FiCheck className="badge-icon" />
                  <span>Active</span>
                </div>
              </div>
            )}
          </div>
        ))}
      </div>

      {/* Mode Preview */}
      <div className="mode-preview-section">
        <h4 className="preview-title">
          <span className="preview-icon">👁️</span>
          Mode Preview
        </h4>
        
        <div className="preview-container">
          <div className="preview-sample">
            <div 
              className="preview-header"
              style={{ 
                backgroundColor: theme.colors.header || theme.colors.primary,
                color: theme.colors.textInverse || 'white'
              }}
            >
              <span className="preview-text">Header</span>
            </div>
            
            <div 
              className="preview-content"
              style={{ 
                backgroundColor: theme.colors.background,
                color: theme.colors.textPrimary
              }}
            >
              <div className="preview-item">
                <div 
                  className="preview-card"
                  style={{ 
                    backgroundColor: theme.colors.surface,
                    border: `1px solid ${theme.colors.border}`
                  }}
                >
                  <span>Sample Card</span>
                </div>
              </div>
              
              <div className="preview-item">
                <button 
                  className="preview-button"
                  style={{ 
                    backgroundColor: theme.colors.button.background,
                    color: theme.colors.button.text,
                    border: `1px solid ${theme.colors.button.border}`
                  }}
                >
                  Sample Button
                </button>
              </div>
              
              <div className="preview-item">
                <div 
                  className="preview-text-sample"
                  style={{ color: theme.colors.textSecondary }}
                >
                  Secondary text for context and descriptions
                </div>
              </div>
            </div>
            
            <div 
              className="preview-footer"
              style={{ 
                backgroundColor: theme.colors.sidebar || theme.colors.surface,
                borderTop: `1px solid ${theme.colors.divider}`
              }}
            >
              <span className="preview-text">Sidebar/Footer</span>
            </div>
          </div>
          
          <div className="preview-info">
            <div className="info-row">
              <span className="info-label">Current Mode:</span>
              <span className="info-value">{currentMode.name}</span>
            </div>
            <div className="info-row">
              <span className="info-label">Theme Type:</span>
              <span className="info-value">{theme.type === 'light' ? 'Light' : 'Dark'}</span>
            </div>
            <div className="info-row">
              <span className="info-label">Primary Color:</span>
              <span className="info-value" style={{ color: theme.colors.primary }}>
                {theme.colors.primary}
              </span>
            </div>
            <div className="info-row">
              <span className="info-label">Text Contrast:</span>
              <span className="info-value">
                {theme.type === 'light' ? 'Dark on Light' : 'Light on Dark'}
              </span>
            </div>
          </div>
        </div>
      </div>

      {/* Usage Tips */}
      <div className="usage-tips">
        <h4 className="tips-title">
          <span className="tips-icon">💡</span>
          Usage Tips
        </h4>
        
        <div className="tips-grid">
          <div className="tip-card">
            <div className="tip-icon">👁️</div>
            <div className="tip-content">
              <h5>Eye Comfort</h5>
              <p>Use Dark mode in low-light environments to reduce eye strain and blue light exposure.</p>
            </div>
          </div>
          
          <div className="tip-card">
            <div className="tip-icon">🔋</div>
            <div className="tip-content">
              <h5>Battery Saving</h5>
              <p>Dark themes can save battery life on OLED and AMOLED displays.</p>
            </div>
          </div>
          
          <div className="tip-card">
            <div className="tip-icon">🔄</div>
            <div className="tip-content">
              <h5>Auto Switching</h5>
              <p>Device mode automatically switches with your system theme for seamless experience.</p>
            </div>
          </div>
          
          <div className="tip-card">
            <div className="tip-icon">🎨</div>
            <div className="tip-content">
              <h5>Brand Colors</h5>
              <p>Custom mode allows you to match the theme with your brand colors.</p>
            </div>
          </div>
        </div>
      </div>
    </div>
  );
};

export default ThemeModeSelector;
