import React, { useEffect, useRef, useState } from "react";
import { FiBell, FiSettings, FiLogOut, FiUser } from "react-icons/fi";
import { useNavigate } from "react-router-dom";
import { useTheme } from "../../contexts/ThemeContext";
import "./Header.css";

const Header = ({
  title = "Flattening Section",
  subtitle = "Wire flattening process management"
}) => {
  const navigate = useNavigate();
  const tickerRef = useRef(null);
  const [isMobile, setIsMobile] = useState(false);
  
  // USE THEME HOOK
  const { theme } = useTheme();

  // Check mobile on mount and resize
  useEffect(() => {
    const checkMobile = () => {
      setIsMobile(window.innerWidth < 768);
    };

    checkMobile();
    window.addEventListener("resize", checkMobile);
    return () => window.removeEventListener("resize", checkMobile);
  }, []);

  // Get user info
  const storedUser = localStorage.getItem("user");
  const user = storedUser
    ? JSON.parse(storedUser)
    : { display_name: "Admin User", email: "admin@example.com", role: "Administrator" };

  const getInitials = (user) => {
    if (!user) return "AU";
    if (user.display_name) {
      const parts = user.display_name.trim().split(" ");
      if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
      return (parts[0][0] + parts[1][0]).toUpperCase();
    }
    return "AU";
  };

  const initials = getInitials(user);

  // Ticker animation - Only on desktop
  useEffect(() => {
    if (isMobile) return;
    
    const tickerElement = tickerRef.current;
    if (!tickerElement) return;

    let animationId;
    let position = 0;
    const speed = 0.8;

    const animateTicker = () => {
      position -= speed;
      if (position <= -400) {
        position = 0;
      }
      tickerElement.style.transform = `translateX(${position}px)`;
      animationId = requestAnimationFrame(animateTicker);
    };

    animationId = requestAnimationFrame(animateTicker);

    return () => {
      if (animationId) {
        cancelAnimationFrame(animationId);
      }
    };
  }, [isMobile]);

  const handleSettingsClick = () => {
    navigate("/settings/theme");
  };

  const handleProfileClick = () => {
    navigate("/profile");
  };

  const handleLogout = () => {
    localStorage.removeItem("token");
    localStorage.removeItem("user");
    localStorage.removeItem("authToken");
    localStorage.removeItem("userData");
    sessionStorage.clear();
    navigate("/login");
  };

  return (
    <div 
      className="header-container"
      style={{
        width: "100%",
        background: `linear-gradient(135deg, ${theme.colors.primary} 0%, ${theme.colors.primaryLight} 50%, ${theme.colors.primaryDark} 100%)`,
        borderBottom: `2px solid ${theme.colors.border}`,
        color: theme.colors.textPrimary,
        // REMOVE ALL MARGINS
        margin: "0",
        padding: "0",
      }}
    >
      {/* Ticker Bar - Only show on desktop */}
      {!isMobile && (
        <div 
          className="ticker-bar"
          style={{
            background: `linear-gradient(90deg, ${theme.colors.primary} 0%, ${theme.colors.secondary} 50%, ${theme.colors.primary} 100%)`,
            borderBottom: `1px solid ${theme.colors.border}`,
          }}
        >
          <div ref={tickerRef} className="ticker-content">
            {[...Array(8)].map((_, i) => (
              <div key={i} className="ticker-item">
                <img
                  src="/images/logoA.png"
                  alt="PWI Logo"
                  className="ticker-logo"
                  style={{
                    filter: theme.type === 'dark' ? 'invert(1) brightness(1.5)' : 'invert(0)',
                  }}
                />
                <span className="ticker-text">
                  Pakistan Wire Industries
                </span>
                <span className="ticker-tag">
                  SPI & CCD
                </span>
                <span className="ticker-separator">|</span>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* MAIN HEADER CONTENT - FIXED FOR BOTH */}
      <div className="header-main-content">
        {/* Left Section - Logo & Title */}
        <div className="left-section">
          <div className="logo-title-container">
            <img
              src="/images/logoB.png"
              alt="Pakistan Wire Industries Logo"
              className="main-logo"
              style={{
                filter: theme.type === 'dark' ? 'invert(1) brightness(1.5)' : 'invert(0)',
              }}
            />
            <div className="title-container">
              <h1 className="main-title">
                {title}
              </h1>
              <p className="main-subtitle">
                {subtitle}
              </p>
            </div>
          </div>
          
          {/* Company Tag - Only desktop */}
          {!isMobile && (
            <div className="company-tag">
              <span className="company-text">
                PWI Pvt Ltd
              </span>
            </div>
          )}
        </div>

        {/* Right Section - Icons & User Info */}
        <div className="right-section">
          {/* All buttons/icons in one line */}
          <div className="header-icons-row">
            {/* Notification Button */}
            <button 
              className="notification-btn"
              title="Notifications"
            >
              <FiBell />
              <span className="notification-badge">3</span>
            </button>

            {/* Settings Button */}
            <button 
              className="settings-btn" 
              onClick={handleSettingsClick}
              title="Theme Settings"
            >
              <FiSettings />
            </button>

            {/* Profile Button */}
            <button 
              className="profile-btn"
              onClick={handleProfileClick}
              title="User Profile"
            >
              <FiUser />
            </button>

            {/* User Info - Compact */}
            <div 
              className="user-section-compact"
              onClick={handleProfileClick}
              title={`${user.display_name} - ${user.email}`}
            >
              <div className="user-avatar-compact">
                {initials}
              </div>
              {!isMobile && (
                <div className="user-info-compact">
                  <div className="user-name-compact">
                    {user.display_name}
                  </div>
                  <div className="user-email-compact">
                    {user.email}
                  </div>
                </div>
              )}
            </div>

            {/* Logout Button */}
            <button 
              className="logout-btn-compact" 
              onClick={handleLogout}
              title="Logout"
            >
              {!isMobile ? (
                <>
                  <FiLogOut />
                  <span>Logout</span>
                </>
              ) : (
                <FiLogOut />
              )}
            </button>
          </div>
        </div>
      </div>
    </div>
  );
};

export default Header;
