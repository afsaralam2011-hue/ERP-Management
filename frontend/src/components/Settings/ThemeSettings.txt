// src/components/Settings/ThemeSettings.jsx
import React, { useState, useEffect } from 'react';
import { useTheme } from '../../contexts/ThemeContext';
import ColorPicker from './ColorPicker';
import './ThemeSettings.css';

const ThemeSettings = () => {
  const {
    currentTheme,
    mode,
    themes,
    customThemes,
    setTheme,
    setMode,
    createCustomTheme,
    deleteCustomTheme,
    exportTheme,
    importTheme,
    resetTheme,
  } = useTheme();

  const [showCustomThemeModal, setShowCustomThemeModal] = useState(false);
  const [customThemeName, setCustomThemeName] = useState('');
  const [customColors, setCustomColors] = useState({
    primary: '#1976D2',
    secondary: '#DC004E',
    background: '#FFFFFF',
    surface: '#F5F5F5',
    textPrimary: '#212121',
    textSecondary: '#757575',
  });
  const [showBenchmarks, setShowBenchmarks] = useState(false);
  const [importingFile, setImportingFile] = useState(null);

  // Handle mode change - یہ function درست کریں
  const handleModeChange = (newMode) => {
    console.log('Changing mode to:', newMode);
    setMode(newMode);
    
    // Force text colors immediately
    setTimeout(() => {
      if (newMode === 'dark') {
        // Force white text for dark theme
        document.body.style.color = '#FFFFFF';
        document.querySelectorAll('h1, h2, h3, h4, h5, h6, p, span, div, a').forEach(el => {
          if (window.getComputedStyle(el).color === 'rgb(33, 33, 33)' || 
              window.getComputedStyle(el).color === '#212121') {
            el.style.color = '#FFFFFF';
          }
        });
      } else {
        // Force black text for light theme
        document.body.style.color = '#212121';
        document.querySelectorAll('h1, h2, h3, h4, h5, h6, p, span, div, a').forEach(el => {
          if (window.getComputedStyle(el).color === 'rgb(255, 255, 255)' || 
              window.getComputedStyle(el).color === '#FFFFFF') {
            el.style.color = '#212121';
          }
        });
      }
    }, 50);
  };

  // Handle theme selection
  const handleThemeSelect = (themeId) => {
    setTheme(themeId);
    
    // Theme change کے بعد text colors کو force کریں
    setTimeout(() => {
      if (mode === 'dark') {
        document.body.style.color = '#FFFFFF';
      }
    }, 100);
  };

  // Create custom theme
  const handleCreateCustomTheme = () => {
    if (!customThemeName.trim()) {
      alert('Please enter a theme name');
      return;
    }

    const newTheme = createCustomTheme(customThemeName, customColors, mode);
    if (newTheme) {
      setTheme(newTheme.id);
      setShowCustomThemeModal(false);
      setCustomThemeName('');
      
      // Custom theme create کرنے کے بعد text colors کو force کریں
      if (mode === 'dark') {
        setTimeout(() => {
          document.body.style.color = '#FFFFFF';
        }, 50);
      }
    }
  };

  // Delete custom theme
  const handleDeleteCustomTheme = (themeId) => {
    if (window.confirm('Are you sure you want to delete this custom theme?')) {
      deleteCustomTheme(themeId);
    }
  };

  // Export theme
  const handleExportTheme = () => {
    exportTheme(currentTheme);
  };

  // Import theme
  const handleImportTheme = (e) => {
    const file = e.target.files[0];
    if (file) {
      setImportingFile(file.name);
      importTheme(file).then((importedTheme) => {
        if (importedTheme) {
          alert(`Theme "${importedTheme.name}" imported successfully!`);
          setTheme(importedTheme.id);
          
          // Import کے بعد text colors کو force کریں
          if (mode === 'dark') {
            setTimeout(() => {
              document.body.style.color = '#FFFFFF';
            }, 50);
          }
        }
        setImportingFile(null);
      }).catch(() => {
        alert('Failed to import theme. Please check the file format.');
        setImportingFile(null);
      });
    }
  };

  // Reset theme
  const handleResetTheme = () => {
    resetTheme();
    
    // Reset کے بعد text colors کو force کریں
    setTimeout(() => {
      document.body.style.color = '#212121'; // Light theme کا default text color
    }, 50);
  };

  // Predefined themes (non-custom)
  const predefinedThemes = themes.filter(t => !t.isCustom);

  // Component mount پر current mode کے مطابق text colors set کریں
  useEffect(() => {
    // Small delay to ensure DOM is ready
    setTimeout(() => {
      if (mode === 'dark') {
        document.body.style.color = '#FFFFFF';
      } else {
        document.body.style.color = '#212121';
      }
    }, 100);
  }, [mode]);

  return (
    <div className="theme-settings">
      {/* Performance Benchmarks Toggle */}
      <div className="benchmarks-toggle-section">
        <button
          className="benchmarks-toggle-button"
          onClick={() => setShowBenchmarks(!showBenchmarks)}
        >
          <span className="toggle-icon">⚡</span>
          {showBenchmarks ? 'Hide Performance Benchmarks' : 'Show Performance Benchmarks'}
          <span className="toggle-arrow">{showBenchmarks ? '▼' : '▶'}</span>
        </button>
      </div>

      {showBenchmarks && (
        <div className="benchmarks-preview">
          <h3>Performance Benchmarks</h3>
          <p>Theme switching: &lt; 200ms target</p>
          <p>Page load: &lt; 1.5s target</p>
          <p>Color picker response: &lt; 50ms target</p>
        </div>
      )}

      {/* Theme Mode Section - Updated Design */}
      <div className="theme-mode-section">
        <h3 className="section-title">
          <span className="section-icon">🎨</span>
          Theme Mode
        </h3>
        <div className="mode-cards">
          <div 
            className={`mode-card ${mode === 'light' ? 'active' : ''}`}
            onClick={() => handleModeChange('light')}
          >
            <div className="mode-card-icon">☀️</div>
            <div className="mode-card-content">
              <h4 className="mode-card-title">Light</h4>
              <p className="mode-card-description">Bright and clean interface</p>
            </div>
            {mode === 'light' && <div className="mode-card-check">✓</div>}
          </div>
          
          <div 
            className={`mode-card ${mode === 'dark' ? 'active' : ''}`}
            onClick={() => handleModeChange('dark')}
          >
            <div className="mode-card-icon">🌙</div>
            <div className="mode-card-content">
              <h4 className="mode-card-title">Dark</h4>
              <p className="mode-card-description">Reduced eye strain</p>
            </div>
            {mode === 'dark' && <div className="mode-card-check">✓</div>}
          </div>
        </div>
      </div>

      {/* Predefined Themes Grid */}
      <div className="theme-selection-section">
        <h3 className="section-title">
          <span className="section-icon">🎭</span>
          Predefined Themes
        </h3>
        <div className="theme-grid modern-grid">
          {predefinedThemes.map((theme) => (
            <div
              key={theme.id}
              className={`theme-card ${currentTheme?.id === theme.id ? 'active' : ''}`}
              onClick={() => handleThemeSelect(theme.id)}
            >
              <div className="theme-card-preview">
                <div className="theme-preview-colors">
                  <div className="color-strip primary" style={{ backgroundColor: theme.colors.primary }} />
                  <div className="color-strip secondary" style={{ backgroundColor: theme.colors.secondary }} />
                  <div className="color-strip background" style={{ backgroundColor: theme.colors.background }} />
                  <div className="color-strip surface" style={{ backgroundColor: theme.colors.surface }} />
                </div>
              </div>
              <div className="theme-card-info">
                <h4 className="theme-card-name">{theme.name}</h4>
                <p className="theme-card-type">Predefined</p>
              </div>
              {currentTheme?.id === theme.id && (
                <div className="theme-card-badge">Active</div>
              )}
            </div>
          ))}
        </div>
      </div>

      {/* Custom Themes */}
      {customThemes.length > 0 && (
        <div className="custom-themes-section">
          <h3 className="section-title">
            <span className="section-icon">✨</span>
            Custom Themes
          </h3>
          <div className="theme-grid modern-grid">
            {customThemes.map((theme) => (
              <div
                key={theme.id}
                className={`theme-card custom ${currentTheme?.id === theme.id ? 'active' : ''}`}
              >
                <div className="theme-card-preview" onClick={() => handleThemeSelect(theme.id)}>
                  <div className="theme-preview-colors">
                    <div className="color-strip primary" style={{ backgroundColor: theme.colors.primary }} />
                    <div className="color-strip secondary" style={{ backgroundColor: theme.colors.secondary }} />
                    <div className="color-strip background" style={{ backgroundColor: theme.colors.background }} />
                    <div className="color-strip surface" style={{ backgroundColor: theme.colors.surface }} />
                  </div>
                </div>
                <div className="theme-card-info">
                  <h4 className="theme-card-name">{theme.name}</h4>
                  <p className="theme-card-type">Custom</p>
                </div>
                {currentTheme?.id === theme.id && (
                  <div className="theme-card-badge">Active</div>
                )}
                <button
                  className="theme-card-delete"
                  onClick={(e) => {
                    e.stopPropagation();
                    handleDeleteCustomTheme(theme.id);
                  }}
                  title="Delete theme"
                >
                  🗑️
                </button>
              </div>
            ))}
          </div>
        </div>
      )}

      {/* Theme Actions - Modern Grid Layout */}
      <div className="theme-actions-section">
        <h3 className="section-title">
          <span className="section-icon">⚡</span>
          Theme Actions
        </h3>
        <div className="actions-grid">
          {/* Create Theme */}
          <div 
            className="action-card create"
            onClick={() => setShowCustomThemeModal(true)}
          >
            <div className="action-icon">🎨</div>
            <div className="action-content">
              <h4 className="action-title">Create Theme</h4>
              <p className="action-description">Design your own colors</p>
            </div>
            <div className="action-arrow">→</div>
          </div>

          {/* Export Theme */}
          <div 
            className="action-card export"
            onClick={handleExportTheme}
          >
            <div className="action-icon">📤</div>
            <div className="action-content">
              <h4 className="action-title">Export Current Theme</h4>
              <p className="action-description">Save as JSON file</p>
            </div>
            <div className="action-arrow">→</div>
          </div>

          {/* Import Theme */}
          <label className="action-card import">
            <div className="action-icon">📥</div>
            <div className="action-content">
              <h4 className="action-title">Import Theme</h4>
              <p className="action-description">
                {importingFile ? `Importing: ${importingFile}` : 'Load from JSON file'}
              </p>
            </div>
            <div className="action-arrow">→</div>
            <input
              type="file"
              accept=".json"
              onChange={handleImportTheme}
              style={{ display: 'none' }}
            />
          </label>

          {/* Reset Theme */}
          <div 
            className="action-card reset"
            onClick={handleResetTheme}
          >
            <div className="action-icon">🔄</div>
            <div className="action-content">
              <h4 className="action-title">Reset to Default</h4>
              <p className="action-description">Restore original theme</p>
            </div>
            <div className="action-arrow">→</div>
          </div>
        </div>
      </div>

      {/* Custom Theme Modal */}
      {showCustomThemeModal && (
        <div className="modal-overlay" onClick={() => setShowCustomThemeModal(false)}>
          <div className="modal-content modern-modal" onClick={(e) => e.stopPropagation()}>
            <div className="modal-header">
              <div className="modal-header-content">
                <span className="modal-icon">🎨</span>
                <h3>Create Custom Theme</h3>
              </div>
              <button 
                className="modal-close"
                onClick={() => setShowCustomThemeModal(false)}
              >
                ✕
              </button>
            </div>
            
            <div className="modal-body">
              <div className="form-group">
                <label>Theme Name</label>
                <input
                  type="text"
                  value={customThemeName}
                  onChange={(e) => setCustomThemeName(e.target.value)}
                  placeholder="Enter theme name (e.g., My Blue Theme)"
                  className="theme-name-input modern-input"
                />
              </div>

              <div className="color-pickers-grid">
                <ColorPicker
                  label="Primary Color"
                  color={customColors.primary}
                  onChange={(color) => setCustomColors({ ...customColors, primary: color })}
                />
                <ColorPicker
                  label="Secondary Color"
                  color={customColors.secondary}
                  onChange={(color) => setCustomColors({ ...customColors, secondary: color })}
                />
                <ColorPicker
                  label="Background"
                  color={customColors.background}
                  onChange={(color) => setCustomColors({ ...customColors, background: color })}
                />
                <ColorPicker
                  label="Surface"
                  color={customColors.surface}
                  onChange={(color) => setCustomColors({ ...customColors, surface: color })}
                />
                <ColorPicker
                  label="Text Primary"
                  color={customColors.textPrimary}
                  onChange={(color) => setCustomColors({ ...customColors, textPrimary: color })}
                />
                <ColorPicker
                  label="Text Secondary"
                  color={customColors.textSecondary}
                  onChange={(color) => setCustomColors({ ...customColors, textSecondary: color })}
                />
              </div>
            </div>

            <div className="modal-footer">
              <button 
                className="modal-button secondary"
                onClick={() => setShowCustomThemeModal(false)}
              >
                Cancel
              </button>
              <button 
                className="modal-button primary"
                onClick={handleCreateCustomTheme}
              >
                Create Theme
              </button>
            </div>
          </div>
        </div>
      )}
    </div>
  );
};

export default ThemeSettings;
