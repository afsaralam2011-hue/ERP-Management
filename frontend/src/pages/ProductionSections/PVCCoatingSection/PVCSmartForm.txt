import React, { useState, useEffect, useMemo, useCallback } from 'react';
import { useNavigate } from 'react-router-dom';
import { 
  FiSave, FiClock, FiCheck, FiAlertCircle,
  FiRefreshCw, FiArrowLeft, FiArrowRight,
  FiCpu, FiChevronRight, FiChevronLeft, 
  FiCalendar, FiX, FiLock,
  FiXCircle, FiUser, FiPackage, FiActivity,
  FiBarChart, FiPlus, FiTrash2,
  FiSettings, FiMaximize2, FiMinimize2, FiInfo,
  FiMenu, FiTrendingUp, FiSkipForward, FiEdit,
  FiEye, FiSun, FiMoon, FiTarget
} from 'react-icons/fi';
import { supabase } from '../../../supabaseClient';

const PVCSmartForm = () => {
  const navigate = useNavigate();
  
  // Constants
  const CURRENT_SECTION = 'PVC';
  
  // States
  const [selectedShift, setSelectedShift] = useState('');
  const [shifts, setShifts] = useState([]);
  const [machineData, setMachineData] = useState({});
  const [items, setItems] = useState([]);
  const [targets, setTargets] = useState([]);
  const [operators, setOperators] = useState([]);
  const [loading, setLoading] = useState(true);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState('');
  const [success, setSuccess] = useState('');
  const [activeShiftIndex, setActiveShiftIndex] = useState(0);
  const [currentUser, setCurrentUser] = useState('');
  const [currentDate, setCurrentDate] = useState('');
  const [duplicateCheck, setDuplicateCheck] = useState({
    checking: false,
    duplicateFound: false,
    existingEntry: null
  });
  
  // Flow States
  const [emptyMachines, setEmptyMachines] = useState([]);
  const [completedMachines, setCompletedMachines] = useState([]);

  // Entry form settings
  const [entryFormExpanded, setEntryFormExpanded] = useState(false);
  const [selectedMachineForEntry, setSelectedMachineForEntry] = useState(null);
  const [showEntryForm, setShowEntryForm] = useState(false);

  // Auto save and move flag
  const [autoMoveEnabled] = useState(true);
  
  // Theme State
  const [currentTheme, setCurrentTheme] = useState('light');
  
  // Side Panel States
  const [showSidePanel, setShowSidePanel] = useState(false);
  const [sidePanelContent, setSidePanelContent] = useState('summary');
  const [currentMachineDetails, setCurrentMachineDetails] = useState(null);

  // ==================== STYLE VARIABLES ====================
  const styles = {
    // Theme based colors
    light: {
      bgPrimary: '#ffffff',
      bgSecondary: '#f8fafc',
      bgTertiary: '#f1f5f9',
      textPrimary: '#1e293b',
      textSecondary: '#64748b',
      textTertiary: '#94a3b8',
      borderColor: '#e2e8f0',
      shadowColor: 'rgba(0, 0, 0, 0.08)',
      cardBg: '#ffffff',
      modalBg: '#ffffff',
      modalOverlay: 'rgba(0, 0, 0, 0.5)',
      headerBg: '#ffffff',
      btnPrimaryBg: '#005461',
      btnPrimaryText: '#ffffff',
      btnPrimaryHover: '#00363d',
      btnSecondaryBg: '#f8fafc',
      btnSecondaryText: '#1e293b',
      btnSecondaryHover: '#f1f5f9',
      success: '#10b981',
      successLight: '#d1fae5',
      successDark: '#059669',
      warning: '#f59e0b',
      warningLight: '#fef3c7',
      warningDark: '#d97706',
      error: '#ef4444',
      errorLight: '#fee2e2',
      errorDark: '#dc2626',
      info: '#3b82f6',
      infoLight: '#dbeafe',
      infoDark: '#1d4ed8',
      primary: '#005461',
      primaryLight: '#018790',
      primaryDark: '#00363d',
      secondary: '#00B7B5',
      secondaryLight: '#00d4d2',
      secondaryDark: '#009b99'
    },
    dark: {
      bgPrimary: '#0f172a',
      bgSecondary: '#1e293b',
      bgTertiary: '#334155',
      textPrimary: '#e2e8f0', // ہلکا سرمئی سفید - ڈارک تھیم کے لیے
      textSecondary: '#cbd5e1', // ہلکا سرمئی
      textTertiary: '#94a3b8', // سرمئی نیلا
      borderColor: '#475569',
      shadowColor: 'rgba(0, 0, 0, 0.3)',
      cardBg: '#1e293b',
      modalBg: '#1e293b',
      modalOverlay: 'rgba(0, 0, 0, 0.7)',
      headerBg: '#1e293b',
      btnPrimaryBg: '#018790',
      btnPrimaryText: '#ffffff',
      btnPrimaryHover: '#005461',
      btnSecondaryBg: '#334155',
      btnSecondaryText: '#e2e8f0', // ڈارک تھیم میں ہلکا ٹیکسٹ
      btnSecondaryHover: '#475569',
      success: '#10b981',
      successLight: '#064e3b',
      successDark: '#047857',
      warning: '#f59e0b',
      warningLight: '#78350f',
      warningDark: '#d97706',
      error: '#ef4444',
      errorLight: '#7f1d1d',
      errorDark: '#dc2626',
      info: '#3b82f6',
      infoLight: '#1e3a8a',
      infoDark: '#1d4ed8',
      primary: '#005461',
      primaryLight: '#018790',
      primaryDark: '#00363d',
      secondary: '#00B7B5',
      secondaryLight: '#00d4d2',
      secondaryDark: '#009b99'
    }
  };

  const themeStyles = currentTheme === 'light' ? styles.light : styles.dark;

  // ==================== INITIALIZE THEME ====================
  useEffect(() => {
    const savedTheme = localStorage.getItem('pwi_theme') || 'light';
    setCurrentTheme(savedTheme);
    document.documentElement.setAttribute('data-theme', savedTheme);
    
    const handleThemeChange = (event) => {
      const newTheme = event.detail.theme;
      setCurrentTheme(newTheme);
      document.documentElement.setAttribute('data-theme', newTheme);
    };
    
    window.addEventListener('theme-change', handleThemeChange);
    return () => window.removeEventListener('theme-change', handleThemeChange);
  }, []);
  
  // ==================== TOGGLE THEME ====================
  const toggleTheme = () => {
    const newTheme = currentTheme === 'light' ? 'dark' : 'light';
    setCurrentTheme(newTheme);
    document.documentElement.setAttribute('data-theme', newTheme);
    localStorage.setItem('pwi_theme', newTheme);
    
    window.dispatchEvent(new CustomEvent('theme-change', {
      detail: { theme: newTheme, timestamp: new Date().toISOString() }
    }));
    
    return newTheme;
  };
  
  // ==================== UTILITY FUNCTIONS ====================
  const formatNumber = useCallback((num) => {
    const number = parseFloat(num) || 0;
    if (Number.isInteger(number)) {
      return number.toLocaleString('en-US');
    }
    const rounded = Math.round(number * 100) / 100;
    return rounded.toLocaleString('en-US', {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2
    });
  }, []);
  
  const getEfficiencyStatus = useCallback((efficiency) => {
    if (efficiency >= 70) {
      return {
        text: efficiency >= 100 ? 'Excellent' : 'Good',
        color: efficiency >= 100 ? '#00ff88' : '#4cc9f0'
      };
    } else {
      return {
        text: 'Below Target',
        color: '#ff4444'
      };
    }
  }, []);
  
  const getShiftDisplayName = useCallback((shiftCode) => {
    const shift = shifts.find(s => s.shift_code === shiftCode);
    if (!shift) return `Shift ${shiftCode}`;
    return shift.shift_name === 'Day Shift' ? 'Day Shift' : 
           shift.shift_name === 'Night Shift' ? 'Night Shift' : 
           `Shift ${shiftCode}`;
  }, [shifts]);
  
  // ==================== GET CURRENT USER AND DATE ====================
  useEffect(() => {
    const getUser = async () => {
      try {
        const { data: { user } } = await supabase.auth.getUser();
        if (user) setCurrentUser(user.email.split('@')[0]);
        else setCurrentUser('admin');
      } catch (err) {
        console.error('User fetch error:', err);
        setCurrentUser('admin');
      }
    };

    const today = new Date();
    setCurrentDate(today.toISOString().split('T')[0]);
    getUser();
  }, []);
  
  // ==================== FETCH DATA ====================
  useEffect(() => {
    const fetchData = async () => {
      try {
        setLoading(true);
        console.log('Fetching data for PVC section...');
        
        // Fetch targets
        const { data: targetsData, error: targetsError } = await supabase
          .from('targets')
          .select('*')
          .eq('section_name', CURRENT_SECTION)
          .eq('is_active', true);

        if (targetsError) {
          console.error('Targets fetch error:', targetsError);
          throw targetsError;
        }

        console.log('Targets loaded:', targetsData?.length || 0);

        // Fetch operators from database
        const { data: operatorsData, error: operatorsError } = await supabase
          .from('operators')
          .select('operator_name')
          .eq('section', CURRENT_SECTION)
          .order('operator_name');

        if (operatorsError) {
          console.error('Operators fetch error:', operatorsError);
          // Don't throw error, continue without operators
        }

        console.log('Operators loaded:', operatorsData?.length || 0);

        // Get unique shift codes from targets
        const shiftCodes = targetsData ? [...new Set(targetsData.map(target => target.shift_code))] : [];
        console.log('Shift codes from targets:', shiftCodes);

        if (shiftCodes.length > 0) {
          // Fetch shifts
          const { data: shiftsData, error: shiftsError } = await supabase
            .from('shifts')
            .select('*')
            .in('shift_code', shiftCodes)
            .order('shift_code');

          if (shiftsError) {
            console.error('Shifts fetch error:', shiftsError);
            throw shiftsError;
          }

          console.log('Shifts loaded:', shiftsData?.length || 0);
          setShifts(shiftsData || []);
        } else {
          console.log('No shift codes found in targets');
          setShifts([]);
        }

        // Fetch items
        const { data: itemsData, error: itemsError } = await supabase
          .from('pvcitem')
          .select('*')
          .order('item_code');

        if (itemsError) {
          console.error('Items fetch error:', itemsError);
          throw itemsError;
        }

        console.log('Items loaded:', itemsData?.length || 0);

        setTargets(targetsData || []);
        setItems(itemsData || []);
        setOperators(operatorsData?.map(op => op.operator_name) || []);

        // Auto-select first shift if available
        if (shifts && shifts.length > 0 && !selectedShift) {
          console.log('Auto-selecting first shift:', shifts[0].shift_code);
          setTimeout(() => {
            handleShiftSelect(shifts[0].shift_code);
            setActiveShiftIndex(0);
          }, 100);
        }

      } catch (error) {
        console.error('Data fetch error:', error);
        setError('Failed to load data. Please try again.');
      } finally {
        setLoading(false);
      }
    };

    fetchData();
  }, [CURRENT_SECTION]);
  
  // ==================== MACHINES FOR CURRENT SHIFT ====================
  const machinesForCurrentShift = useMemo(() => {
    if (!selectedShift) {
      console.log('No shift selected');
      return [];
    }
    
    const machineSet = new Set();
    const machinesList = [];
    
    targets.forEach(target => {
      if (target.shift_code === selectedShift && 
          target.section_name === CURRENT_SECTION &&
          target.machine_no && 
          !machineSet.has(target.machine_no)) {
        
        machineSet.add(target.machine_no);
        machinesList.push(target.machine_no);
      }
    });

    machinesList.sort((a, b) => {
      const numA = parseInt(a.replace(/\D/g, '')) || 0;
      const numB = parseInt(b.replace(/\D/g, '')) || 0;
      return numA - numB;
    });

    console.log('Machines found:', machinesList.length);
    return machinesList;
  }, [selectedShift, targets]);
  
  // ==================== CALCULATIONS ====================
  const calculateMachineProductionTotal = useCallback((machineNo) => {
    const machine = machineData[machineNo];
    if (!machine || !machine.items) return 0;
    return machine.items.reduce((total, item) => total + (parseFloat(item.production_quantity) || 0), 0);
  }, [machineData]);

  const calculateMachineWeightTotal = useCallback((machineNo) => {
    const machine = machineData[machineNo];
    if (!machine || !machine.items) return 0;
    return machine.items.reduce((total, item) => total + (parseFloat(item.weight) || 0), 0);
  }, [machineData]);

  // ==================== SECTION TOTALS ====================
  const sectionProductionTotal = useMemo(() => {
    return Object.keys(machineData).reduce((total, machineNo) => {
      return total + calculateMachineProductionTotal(machineNo);
    }, 0);
  }, [machineData, calculateMachineProductionTotal]);

  const sectionWeightTotal = useMemo(() => {
    return Object.keys(machineData).reduce((total, machineNo) => {
      return total + calculateMachineWeightTotal(machineNo);
    }, 0);
  }, [machineData, calculateMachineWeightTotal]);

  const totalTarget = useMemo(() => {
    if (!selectedShift) return 0;
    return targets
      .filter(target => target.shift_code === selectedShift && target.section_name === CURRENT_SECTION)
      .reduce((total, target) => total + (parseFloat(target.target_qty) || 0), 0);
  }, [selectedShift, targets]);

  // ==================== UPDATE CURRENT MACHINE DETAILS ====================
  useEffect(() => {
    if (selectedMachineForEntry && machineData[selectedMachineForEntry]) {
      const machine = machineData[selectedMachineForEntry];
      const production = calculateMachineProductionTotal(selectedMachineForEntry);
      const weight = calculateMachineWeightTotal(selectedMachineForEntry);
      const target = machine.target_qty || 0;
      const efficiency = target > 0 ? (production / target) * 100 : 0;
      
      setCurrentMachineDetails({
        machineNo: selectedMachineForEntry,
        production,
        weight,
        target,
        efficiency: parseFloat(efficiency.toFixed(1)),
        operator: machine.operator_name || currentUser,
        itemsCount: machine.items?.length || 0,
        status: machine.isCompleted ? 'Completed' : 'In Progress',
        lastUpdated: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
      });
    }
  }, [selectedMachineForEntry, machineData, currentUser, calculateMachineProductionTotal, calculateMachineWeightTotal]);
  
  // ==================== GET SIDE PANEL SUMMARY ====================
  const getSidePanelSummary = useMemo(() => {
    if (!selectedShift) return null;
    
    const totalMachines = machinesForCurrentShift.length;
    const completed = completedMachines.length;
    const totalProduction = sectionProductionTotal;
    const totalWeight = sectionWeightTotal;
    const totalTargetValue = totalTarget;
    
    const overallEfficiency = totalTargetValue > 0 ? (totalProduction / totalTargetValue) * 100 : 0;
    
    return {
      totalMachines,
      completed,
      pending: totalMachines - completed,
      totalProduction: formatNumber(totalProduction),
      totalWeight: formatNumber(totalWeight),
      totalTarget: formatNumber(totalTargetValue),
      overallEfficiency: parseFloat(overallEfficiency.toFixed(1)),
      shiftName: getShiftDisplayName(selectedShift),
      date: currentDate
    };
  }, [selectedShift, machinesForCurrentShift, completedMachines, sectionProductionTotal, sectionWeightTotal, totalTarget, currentDate, formatNumber, getShiftDisplayName]);
  
  // ==================== OPEN SIDE PANEL ====================
  const openSidePanel = (contentType = 'summary', machineNo = null) => {
    if (machineNo) setSelectedMachineForEntry(machineNo);
    setSidePanelContent(contentType);
    setShowSidePanel(true);
  };
  
  // ==================== CLOSE SIDE PANEL ====================
  const closeSidePanel = () => setShowSidePanel(false);
  
  // ==================== ENHANCED DUPLICATE CHECK ====================
  const checkDuplicateEntry = useCallback(async (shiftCode, date = currentDate) => {
    if (!shiftCode || !date) {
      console.log('Missing shift code or date for duplicate check');
      return false;
    }
    
    try {
      console.log('Checking duplicate for shift:', shiftCode, 'date:', date);
      setDuplicateCheck(prev => ({ ...prev, checking: true }));
      
      const { data: existingEntries, error: checkError } = await supabase
        .from('pvcsection')
        .select('id, shift_code, entry_date, created_at')
        .eq('shift_code', shiftCode)
        .eq('section_name', CURRENT_SECTION)
        .eq('entry_date', date)
        .limit(1);
      
      if (checkError) {
        console.error('Duplicate check error:', checkError);
        return false;
      }
      
      const hasDuplicates = existingEntries && existingEntries.length > 0;
      console.log('Duplicate check result:', hasDuplicates ? 'Duplicate found' : 'No duplicate');
      
      setDuplicateCheck({
        checking: false,
        duplicateFound: hasDuplicates,
        existingEntry: hasDuplicates ? existingEntries[0] : null
      });
      
      return hasDuplicates;
    } catch (error) {
      console.error('Duplicate check failed:', error);
      setDuplicateCheck(prev => ({ ...prev, checking: false }));
      return false;
    }
  }, [currentDate]);

  // ==================== CHECK DUPLICATE ON SHIFT CHANGE ====================
  useEffect(() => {
    if (selectedShift && currentDate) {
      console.log('Checking duplicate for selected shift:', selectedShift);
      checkDuplicateEntry(selectedShift);
    }
  }, [selectedShift, currentDate, checkDuplicateEntry]);
  
  // ==================== GET MACHINES FOR SHIFT ====================
  const getMachinesForShift = useCallback((shiftCode) => {
    if (!shiftCode) {
      console.log('No shift code provided');
      return [];
    }
    
    console.log('Getting machines for shift:', shiftCode);
    
    const machineSet = new Set();
    const machinesForShift = [];
    
    targets.forEach(target => {
      if (target.shift_code === shiftCode && 
          target.section_name === CURRENT_SECTION &&
          target.machine_no) {
        
        if (!machineSet.has(target.machine_no)) {
          machineSet.add(target.machine_no);
          machinesForShift.push({
            machine_no: target.machine_no,
            machine_id: target.machine_id,
            target: target
          });
        }
      }
    });

    machinesForShift.sort((a, b) => {
      const numA = parseInt(a.machine_no.replace(/\D/g, '')) || 0;
      const numB = parseInt(b.machine_no.replace(/\D/g, '')) || 0;
      return numA - numB;
    });

    console.log('Machines for shift', shiftCode, ':', machinesForShift.length);
    return machinesForShift;
  }, [targets]);
  
  // ==================== ADD NEW OPERATOR ====================
  const addNewOperator = async (operatorName) => {
    if (!operatorName.trim()) return;
    
    try {
      const { error } = await supabase
        .from('operators')
        .insert([{ 
          operator_name: operatorName,
          section: CURRENT_SECTION,
          created_at: new Date().toISOString()
        }]);
      
      if (!error) {
        setOperators(prev => [...prev, operatorName]);
        return true;
      }
      return false;
    } catch (err) {
      console.error('Failed to add operator:', err);
      return false;
    }
  };

  // ==================== HANDLE SHIFT SELECTION ====================
  const handleShiftSelect = async (shiftCode) => {
    console.log('Handling shift selection:', shiftCode);
    
    if (!shiftCode) {
      console.log('No shift code provided');
      return;
    }

    setSelectedShift(shiftCode);
    setError('');
    setSuccess('');
    setCompletedMachines([]);
    setEmptyMachines([]);
    setShowEntryForm(false);
    setSelectedMachineForEntry(null);
    setShowSidePanel(false);

    const selectedShiftData = shifts.find(s => s.shift_code === shiftCode);
    console.log('Selected shift data:', selectedShiftData);

    const hasDuplicate = await checkDuplicateEntry(shiftCode);
    if (hasDuplicate) {
      setError(`This shift already has data for today (${currentDate}). Please edit existing entry instead.`);
      console.log('Duplicate found, stopping initialization');
      return;
    }

    const draftKey = `pvc_draft_${shiftCode}_${currentDate}`;
    const draft = localStorage.getItem(draftKey);
    console.log('Draft key:', draftKey, 'Exists:', !!draft);

    const initialMachineData = {};
    const machinesForThisShift = getMachinesForShift(shiftCode);
    console.log('Machines for this shift:', machinesForThisShift.length);

    if (machinesForThisShift.length === 0) {
      console.log('No machines found for this shift');
      setMachineData({});
      return;
    }

    machinesForThisShift.forEach(machine => {
      const target = machine.target;
      console.log('Initializing machine:', machine.machine_no, 'target:', target);
      
      initialMachineData[machine.machine_no] = {
        machine_id: target.machine_id || '',
        machine_no: machine.machine_no,
        targets_id: target.targets_id || '',
        target_qty: parseFloat(target.target_qty) || 0,
        unit: target.uom || 'Meter',
        shift_code: shiftCode,
        shift_name: selectedShiftData?.shift_name || shiftCode,
        items: [{ 
          id: Date.now(), 
          item_code: '', 
          item_name: '', 
          raw_material_Spiralsize: '',
          material_type: '',
          finishedproductname: '',
          production_quantity: '', 
          per_meter_wt: '',
          weight: '',
          unit: 'Kg', 
          efficiency: 0
        }],
        operator_name: '',
        users_name: currentUser,
        remarks: '',
        section_name: CURRENT_SECTION,
        entry_date: currentDate,
        isCompleted: false
      };
    });

    console.log('Initial machine data created:', Object.keys(initialMachineData).length, 'machines');
    setMachineData(initialMachineData);
    setEmptyMachines(machinesForThisShift.map(m => m.machine_no));
    
    if (draft) {
      try {
        const parsedDraft = JSON.parse(draft);
        if (parsedDraft.machineData) {
          console.log('Loading draft data');
          setMachineData(parsedDraft.machineData);
          setSuccess('Previous draft loaded');
          setTimeout(() => setSuccess(''), 3000);
          
          const completed = [];
          Object.keys(parsedDraft.machineData).forEach(machineNo => {
            const machine = parsedDraft.machineData[machineNo];
            if (machine.items?.some(item => item.item_code) && machine.operator_name) {
              completed.push(machineNo);
            }
          });
          setCompletedMachines(completed);
          setEmptyMachines(prev => prev.filter(m => !completed.includes(m)));
          console.log('Draft loaded, completed machines:', completed.length);
        }
      } catch (err) {
        console.error('Draft load error:', err);
      }
    }
  };
  
  // ==================== OPEN ENTRY FORM POPUP ====================
  const openEntryFormPopup = (machineNo) => {
    console.log('Opening entry form for machine:', machineNo);
    setSelectedMachineForEntry(machineNo);
    setShowEntryForm(true);
  };

  // ==================== CLOSE ENTRY FORM POPUP ====================
  const closeEntryFormPopup = () => {
    console.log('Closing entry form');
    setShowEntryForm(false);
    setSelectedMachineForEntry(null);
    setEntryFormExpanded(false);
  };

  // ==================== SAVE + CLEAR + NEXT SHIFT FUNCTION ====================
  const saveClearAndNextShift = async () => {
    if (!selectedShift) {
      setError('Please select a shift first');
      return;
    }

    if (machinesForCurrentShift.length === 0) {
      setError('No machines found for this shift');
      return;
    }

    const allMachinesHaveData = machinesForCurrentShift.every(machineNo => {
      const machine = machineData[machineNo];
      return machine?.items?.some(item => 
        item.item_code && item.item_code.trim() !== '' && 
        item.production_quantity && parseFloat(item.production_quantity) > 0
      );
    });

    if (!allMachinesHaveData) {
      setError('Please enter production quantity for at least one item in each machine');
      return;
    }

    const hasDuplicate = await checkDuplicateEntry(selectedShift);
    if (hasDuplicate) {
      setError(`Cannot save: This shift already has data for today (${currentDate}).`);
      return;
    }

    setSaving(true);
    setError('');
    setSuccess('');

    try {
      const allRecords = [];

      machinesForCurrentShift.forEach(machineNo => {
        const machine = machineData[machineNo];
        if (!machine) return;

        machine.items.forEach(item => {
          if (item.item_code && item.item_code.trim() !== '') {
            const selectedItem = items.find(i => i.item_code === item.item_code);
            const productionQty = parseFloat(item.production_quantity) || 0;
            const targetQty = machine.target_qty || 0;
            const itemEfficiency = targetQty > 0 ? (productionQty / targetQty) * 100 : 0;

            allRecords.push({
              section_name: CURRENT_SECTION,
              machine_id: machine.machine_id,
              machine_no: machine.machine_no,
              item_code: item.item_code,
              item_name: selectedItem?.item_name || item.item_name || '',
              raw_material_Spiralsize: item.raw_material_Spiralsize || '',
              material_type: item.material_type || '',
              finishedproductname: item.finishedproductname || '',
              operator_name: machine.operator_name.trim(),
              production_quantity: productionQty,
              per_meter_wt: parseFloat(item.per_meter_wt) || 0,
              weight: parseFloat(item.weight) || 0,
              unit: item.unit || 'Kg',
              efficiency: parseFloat(itemEfficiency.toFixed(1)),
              users_name: machine.users_name || currentUser,
              shift_code: machine.shift_code,
              shift_name: machine.shift_name,
              target_qty: machine.target_qty || 0,
              remarks: machine.remarks || '',
              created_at: new Date().toISOString(),
              updated_at: new Date().toISOString(),

                                  </span>
                                </div>
                              </div>
                            )}
                          </div>
                        )}
                      </div>
                    </div>
                  ))}
                </div>
              </div>

              <div style={{
                display: 'grid',
                gridTemplateColumns: 'repeat(2, 1fr)',
                gap: '12px',
                marginBottom: '20px',
                padding: '16px',
                background: themeStyles.bgSecondary,
                borderRadius: '6px',
                boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
              }}>
                <div style={{ display: 'flex', flexDirection: 'column', gap: '5px' }}>
                  <label style={{
                    color: themeStyles.textSecondary,
                    fontWeight: '600',
                    fontSize: '12px',
                    display: 'flex',
                    alignItems: 'center',
                    gap: '5px'
                  }}>
                    <FiUser /> Operator Name *
                  </label>
                  <div style={{ position: 'relative' }}>
                    <select 
                      value={machineData[selectedMachineForEntry]?.operator_name || ''} 
                      onChange={(e) => {
                        const value = e.target.value;
                        if (value === 'add_new') {
                          const newName = prompt('Enter new operator name:');
                          if (newName && newName.trim()) {
                            addNewOperator(newName.trim()).then(success => {
                              if (success) {
                                setMachineData(prev => ({
                                  ...prev,
                                  [selectedMachineForEntry]: {
                                    ...prev[selectedMachineForEntry],
                                    operator_name: newName.trim()
                                  }
                                }));
                              }
                            });
                          }
                        } else {
                          setMachineData(prev => ({
                            ...prev,
                            [selectedMachineForEntry]: {
                              ...prev[selectedMachineForEntry],
                              operator_name: value
                            }
                          }));
                        }
                      }}
                      style={{
                        padding: '8px 10px',
                        background: themeStyles.bgPrimary,
                        color: themeStyles.textPrimary,
                        border: `1px solid ${themeStyles.borderColor}`,
                        borderRadius: '6px',
                        fontSize: '13px',
                        transition: 'all 0.2s ease',
                        fontFamily: "'Inter', sans-serif",
                        width: '100%'
                      }}
                    >
                      <option value="">-- Select Operator --</option>
                      {operators.map((op, index) => (
                        <option key={index} value={op}>{op}</option>
                      ))}
                      <option value="add_new" style={{ color: themeStyles.primary, fontWeight: 'bold' }}>
                        + Add New Operator
                      </option>
                    </select>
                  </div>
                </div>

                <div style={{ display: 'flex', flexDirection: 'column', gap: '5px' }}>
                  <label style={{
                    color: themeStyles.textSecondary,
                    fontWeight: '600',
                    fontSize: '12px',
                    display: 'flex',
                    alignItems: 'center',
                    gap: '5px'
                  }}>
                    <FiInfo /> Remarks
                  </label>
                  <textarea 
                    value={machineData[selectedMachineForEntry]?.remarks || ''} 
                    onChange={(e) => setMachineData(prev => ({ 
                      ...prev, 
                      [selectedMachineForEntry]: { 
                        ...prev[selectedMachineForEntry], 
                        remarks: e.target.value 
                      } 
                    }))}
                    style={{
                      padding: '8px 10px',
                      background: themeStyles.bgPrimary,
                      color: themeStyles.textPrimary,
                      border: `1px solid ${themeStyles.borderColor}`,
                      borderRadius: '6px',
                      fontSize: '13px',
                      transition: 'all 0.2s ease',
                      fontFamily: "'Inter', sans-serif",
                      resize: 'vertical',
                      minHeight: '70px'
                    }}
                    placeholder="Enter remarks if needed"
                    rows="3"
                  />
                </div>
              </div>

              <div style={{
                display: 'flex',
                justifyContent: 'space-between',
                alignItems: 'center',
                paddingTop: '16px',
                borderTop: `1px solid ${themeStyles.borderColor}`
              }}>
                <button 
                  type="button" 
                  onClick={() => clearMachineData(selectedMachineForEntry)}
                  style={{
                    background: themeStyles.errorLight,
                    color: themeStyles.error,
                    border: '0',
                    borderRadius: '6px',
                    padding: '8px 16px',
                    cursor: 'pointer',
                    fontWeight: '600',
                    transition: 'all 0.2s ease',
                    display: 'flex',
                    alignItems: 'center',
                    gap: '5px',
                    fontSize: '12px',
                    boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                  }}
                >
                  <FiXCircle /> Clear Data
                </button>
                <div style={{ display: 'flex', gap: '10px' }}>
                  <button 
                    type="button" 
                    onClick={closeEntryFormPopup}
                    style={{
                      background: themeStyles.bgTertiary,
                      color: themeStyles.textSecondary,
                      border: '0',
                      borderRadius: '6px',
                      padding: '8px 16px',
                      cursor: 'pointer',
                      fontWeight: '600',
                      transition: 'all 0.2s ease',
                      fontSize: '12px',
                      boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                    }}
                  >
                    Cancel
                  </button>
                  <button 
                    type="button" 
                    onClick={() => {
                      const isCurrentCompleted = machineData[selectedMachineForEntry]?.items?.some(item => item.item_code && item.item_code.trim() !== '' && item.production_quantity && parseFloat(item.production_quantity) > 0);
                      if (isCurrentCompleted) {
                        const currentIndex = machinesForCurrentShift.indexOf(selectedMachineForEntry);
                        let nextMachineIndex = -1;
                        for (let i = currentIndex + 1; i < machinesForCurrentShift.length; i++) {
                          const nextMachine = machineData[machinesForCurrentShift[i]];
                          const isNextCompleted = nextMachine?.items?.some(item => item.item_code && item.item_code.trim() !== '' && item.production_quantity && parseFloat(item.production_quantity) > 0);
                          if (!isNextCompleted) { nextMachineIndex = i; break; }
                        }
                        if (nextMachineIndex !== -1) openEntryFormPopup(machinesForCurrentShift[nextMachineIndex]);
                        else { closeEntryFormPopup(); setSuccess('All machines completed!'); }
                      } else setError('Please complete current machine before moving to next');
                    }}
                    style={{
                      background: `linear-gradient(135deg, #8b5cf6, #7c3aed)`,
                      color: 'white',
                      border: '0',
                      borderRadius: '6px',
                      padding: '8px 16px',
                      cursor: 'pointer',
                      fontWeight: '600',
                      transition: 'all 0.2s ease',
                      fontSize: '12px',
                      boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                    }}
                  >
                    Save & Next Machine
                  </button>
                </div>
              </div>
            </div>
          </div>
        </div>
      )}

      {/* SIDE PANEL */}
      {showSidePanel && (
        <>
          <div 
            onClick={closeSidePanel}
            style={{
              position: 'fixed',
              top: '0',
              left: '0',
              right: '0',
              bottom: '0',
              background: themeStyles.modalOverlay,
              backdropFilter: 'blur(3px)',
              zIndex: '999',
              transition: 'all 0.2s ease'
            }}
          />
          <div style={{
            position: 'fixed',
            top: '0',
            right: '0',
            width: '320px',
            height: '100vh',
            background: themeStyles.bgSecondary,
            boxShadow: `-2px 0 8px ${themeStyles.shadowColor}`,
            zIndex: '1000',
            transition: 'right 0.2s ease',
            display: 'flex',
            flexDirection: 'column'
          }}>
            <div style={{
              background: `linear-gradient(135deg, ${themeStyles.primary}, ${themeStyles.primaryDark})`,
              color: 'white',
              padding: '16px 20px',
              display: 'flex',
              justifyContent: 'space-between',
              alignItems: 'center'
            }}>
              <div style={{
                fontSize: '15px',
                fontWeight: '700',
                display: 'flex',
                alignItems: 'center',
                gap: '6px'
              }}>
                <FiEye /> Production Summary
              </div>
              <button 
                onClick={closeSidePanel}
                style={{
                  background: 'rgba(255, 255, 255, 0.15)',
                  border: '0',
                  borderRadius: '6px',
                  width: '32px',
                  height: '32px',
                  display: 'flex',
                  alignItems: 'center',
                  justifyContent: 'center',
                  cursor: 'pointer',
                  color: 'white',
                  transition: 'all 0.2s ease'
                }}
              >
                <FiX />
              </button>
            </div>
            <div style={{
              flex: '1',
              padding: '16px 20px',
              overflowY: 'auto',
              display: 'flex',
              flexDirection: 'column',
              gap: '16px'
            }}>
              {currentMachineDetails && (
                <div style={{
                  background: themeStyles.bgPrimary,
                  borderRadius: '6px',
                  padding: '12px',
                  boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                }}>
                  <h4 style={{
                    color: themeStyles.textPrimary,
                    fontSize: '13px',
                    marginBottom: '12px',
                    display: 'flex',
                    alignItems: 'center',
                    gap: '6px'
                  }}>
                    <FiCpu /> Machine {currentMachineDetails.machineNo}
                  </h4>
                  <div style={{
                    display: 'grid',
                    gridTemplateColumns: 'repeat(2, 1fr)',
                    gap: '10px'
                  }}>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Production</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{formatNumber(currentMachineDetails.production)} M</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Weight</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{formatNumber(currentMachineDetails.weight)} Kg</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Target</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{formatNumber(currentMachineDetails.target)} M</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Efficiency</span>
                      <span style={{
                        fontWeight: '800',
                        fontSize: '14px',
                        color: getEfficiencyStatus(currentMachineDetails.efficiency).color
                      }}>
                        {currentMachineDetails.efficiency}%
                      </span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Status</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{currentMachineDetails.status}</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Items</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{currentMachineDetails.itemsCount}</span>
                    </div>
                  </div>
                  <div style={{ marginTop: '15px', display: 'flex', flexDirection: 'column', gap: '3px' }}>
                    <span style={{
                      color: themeStyles.textSecondary,
                      fontSize: '10px',
                      fontWeight: '600',
                      textTransform: 'uppercase',
                      letterSpacing: '0.3px'
                    }}>Operator</span>
                    <span style={{
                      fontWeight: '700',
                      fontSize: '13px',
                      color: themeStyles.textPrimary
                    }}>{currentMachineDetails.operator}</span>
                  </div>
                </div>
              )}
              
              {getSidePanelSummary && (
                <div style={{
                  background: themeStyles.bgPrimary,
                  borderRadius: '6px',
                  padding: '12px',
                  boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                }}>
                  <h4 style={{
                    color: themeStyles.textPrimary,
                    fontSize: '13px',
                    marginBottom: '12px',
                    display: 'flex',
                    alignItems: 'center',
                    gap: '6px'
                  }}>
                    <FiBarChart /> Shift Summary
                  </h4>
                  <div style={{
                    display: 'grid',
                    gridTemplateColumns: 'repeat(2, 1fr)',
                    gap: '10px'
                  }}>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Machines</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{getSidePanelSummary.completed}/{getSidePanelSummary.totalMachines}</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Production</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{getSidePanelSummary.totalProduction} M</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Weight</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{getSidePanelSummary.totalWeight} Kg</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Target</span>
                      <span style={{
                        fontWeight: '700',
                        fontSize: '13px',
                        color: themeStyles.textPrimary
                      }}>{getSidePanelSummary.totalTarget} M</span>
                    </div>
                    <div style={{ display: 'flex', flexDirection: 'column', gap: '3px' }}>
                      <span style={{
                        color: themeStyles.textSecondary,
                        fontSize: '10px',
                        fontWeight: '600',
                        textTransform: 'uppercase',
                        letterSpacing: '0.3px'
                      }}>Efficiency</span>
                      <span style={{
                        fontWeight: '800',
                        fontSize: '14px',
                        color: getEfficiencyStatus(getSidePanelSummary.overallEfficiency).color
                      }}>
                        {getSidePanelSummary.overallEfficiency}%
                      </span>
                    </div>
                  </div>
                </div>
              )}
              
              <div style={{
                background: themeStyles.bgPrimary,
                borderRadius: '6px',
                padding: '12px',
                boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
              }}>
                <h4 style={{
                  color: themeStyles.textPrimary,
                  fontSize: '13px',
                  marginBottom: '12px',
                  display: 'flex',
                  alignItems: 'center',
                  gap: '6px'
                }}>
                  <FiTrendingUp /> Efficiency Progress
                </h4>
                <div style={{
                  height: '20px',
                  background: themeStyles.bgTertiary,
                  borderRadius: '10px',
                  overflow: 'hidden',
                  position: 'relative'
                }}>
                  <div style={{
                    height: '100%',
                    background: `linear-gradient(90deg, ${themeStyles.secondary}, ${themeStyles.secondaryLight})`,
                    borderRadius: '10px',
                    transition: 'width 0.5s ease',
                    width: `${Math.min(getSidePanelSummary?.overallEfficiency || 0, 100)}%`
                  }} />
                </div>
                <div style={{
                  display: 'flex',
                  justifyContent: 'space-between',
                  marginTop: '10px'
                }}>
                  <span style={{
                    color: themeStyles.textSecondary,
                    fontSize: '10px',
                    fontWeight: '600'
                  }}>0%</span>
                  <span style={{
                    color: themeStyles.textSecondary,
                    fontSize: '10px',
                    fontWeight: '600'
                  }}>50%</span>
                  <span style={{
                    color: themeStyles.textSecondary,
                    fontSize: '10px',
                    fontWeight: '600'
                  }}>100%</span>
                </div>
              </div>
              
              <div style={{
                background: themeStyles.bgPrimary,
                borderRadius: '6px',
                padding: '12px',
                boxShadow: `0 1px 2px ${themeStyles.shadowColor}`,
                textAlign: 'center'
              }}>
                <h4 style={{
                  color: themeStyles.textPrimary,
                  fontSize: '13px',
                  marginBottom: '12px',
                  display: 'flex',
                  alignItems: 'center',
                  justifyContent: 'center',
                  gap: '6px'
                }}>
                  {currentTheme === 'light' ? <FiSun /> : <FiMoon />} Theme
                </h4>
                <button 
                  onClick={toggleTheme}
                  style={{
                    width: '100%',
                    background: themeStyles.btnSecondaryBg,
                    color: themeStyles.btnSecondaryText,
                    border: '0',
                    borderRadius: '6px',
                    padding: '10px',
                    cursor: 'pointer',
                    fontWeight: '600',
                    fontSize: '12px',
                    transition: 'all 0.2s ease',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    gap: '6px',
                    boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                  }}
                >
                  <span style={{ fontSize: '14px' }}>
                    {currentTheme === 'light' ? <FiMoon /> : <FiSun />}
                  </span>
                  <span style={{ fontSize: '12px' }}>
                    Switch to {currentTheme === 'light' ? 'Dark' : 'Light'} Theme
                  </span>
                </button>
              </div>
              
              <div style={{
                background: themeStyles.bgPrimary,
                borderRadius: '6px',
                padding: '12px',
                boxShadow: `0 1px 2px ${themeStyles.shadowColor}`,
                display: 'grid',
                gridTemplateColumns: 'repeat(2, 1fr)',
                gap: '8px'
              }}>
                <button 
                  onClick={() => selectedMachineForEntry && clearMachineData(selectedMachineForEntry)}
                  style={{
                    background: themeStyles.bgPrimary,
                    border: '0',
                    borderRadius: '6px',
                    padding: '8px',
                    cursor: 'pointer',
                    fontWeight: '600',
                    fontSize: '11px',
                    color: themeStyles.textSecondary,
                    transition: 'all 0.2s ease',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    gap: '5px',
                    boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                  }}
                >
                  <FiXCircle /> Clear Machine
                </button>
                <button 
                  onClick={() => selectedMachineForEntry && openEntryFormPopup(selectedMachineForEntry)}
                  style={{
                    background: themeStyles.bgPrimary,
                    border: '0',
                    borderRadius: '6px',
                    padding: '8px',
                    cursor: 'pointer',
                    fontWeight: '600',
                    fontSize: '11px',
                    color: themeStyles.textSecondary,
                    transition: 'all 0.2s ease',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    gap: '5px',
                    boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                  }}
                >
                  <FiEdit /> Edit Details
                </button>
                <button 
                  onClick={saveClearAndNextShift} 
                  disabled={!checkAllMachinesCompleted()}
                  style={{
                    background: themeStyles.bgPrimary,
                    border: '0',
                    borderRadius: '6px',
                    padding: '8px',
                    cursor: !checkAllMachinesCompleted() ? 'not-allowed' : 'pointer',
                    fontWeight: '600',
                    fontSize: '11px',
                    color: !checkAllMachinesCompleted() ? themeStyles.textTertiary : themeStyles.textSecondary,
                    transition: 'all 0.2s ease',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    gap: '5px',
                    boxShadow: `0 1px 2px ${themeStyles.shadowColor}`,
                    opacity: !checkAllMachinesCompleted() ? '0.5' : '1'
                  }}
                >
                  <FiSave /> Save All
                </button>
                <button 
                  onClick={() => { closeSidePanel(); nextShift(); }}
                  style={{
                    background: themeStyles.bgPrimary,
                    border: '0',
                    borderRadius: '6px',
                    padding: '8px',
                    cursor: 'pointer',
                    fontWeight: '600',
                    fontSize: '11px',
                    color: themeStyles.textSecondary,
                    transition: 'all 0.2s ease',
                    display: 'flex',
                    alignItems: 'center',
                    justifyContent: 'center',
                    gap: '5px',
                    boxShadow: `0 1px 2px ${themeStyles.shadowColor}`
                  }}
                >
                  <FiSkipForward /> Next Shift
                </button>
              </div>
            </div>
          </div>
        </>
      )}
    </div>
  );
};

export default PVCSmartForm;
